# Rule move to spam or trash: implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A tagging rule can carry `"move": "spam" | "trash"`, and the `post-new` hook moves matching new mail into each message's own account's folder for that role.

**Architecture:** One optional field in `rules.json`, known to both readers (`src/tagrules.cpp`, `assets/hooks/mailrules.py`). The hook gains a move step after the tag stages: rename into `<account>/<folder>/cur/` under a fresh name, one `notmuch new --no-hooks` to record the renames, then tag. The dialog gains a *Move to* combo. Spec: `docs/superpowers/specs/2026-11-16-rule-move-design.md`.

**Tech Stack:** Python 3 stdlib (hook), Qt 6 / C++ (`TagRules`, `TagRulesDialog`), notmuch CLI.

```python
# The folders a rule may MOVE mail into, by role. A rule names a role, never a
# folder: a folder name that does not exist is created by the move and reaches
# the server (item 103), and a role resolves per account, so one rule serves
# mail from every account.
MOVE_ROLES = ("spam", "trash")


def role_folders(role, path=None):
    """Every account's (maildir, folder) pair for a move role.

    An account contributes nothing unless it names both its maildir and a
    folder for the role. `spam` and `trash` are mandatory to the application,
    which warns about a missing one; here a missing one just means that
    account's mail is not moved.
    """
    if path is None:
        path = default_path()

    pairs = []
    for _key, section in _accounts(path):
        maildir = section.get("maildir", "").strip()
        folder = section.get(role, "").strip()
        if maildir and folder:
            pairs.append((maildir, folder))
    return pairs
```

```python
_counter = itertools.count(1)


def fresh_name(old_name):
    """A fresh, unique Maildir filename keeping `old_name`'s flags suffix.

    Port of MaildirName::fresh() in the application, and for its reason: the
    `,U=<n>` infix mbsync writes is meaningful only within one folder, and
    carrying it across a folder boundary produced `Maildir error: duplicate
    UID` on real mail (item 162). A file coming from new/ has no `:2,` yet;
    it gets an empty one, since it is going into cur/.
    """
    sep = old_name.find(":2,")
    info = old_name[sep:] if sep >= 0 else ":2,"
    host = (socket.gethostname() or "localhost").replace("/", "_") \
        .replace(":", "_")
    now = time.time()
    return (f"{int(now)}.M{int(now * 1000) % 1000}P{os.getpid()}"
            f"Q{next(_counter)}.{host}{info}")


def account_of(path, root, pairs):
    """The (maildir, folder) pair whose account holds `path`, or None.

    Longest maildir wins, so an account nested inside another's directory
    resolves to itself. Compared as paths, never with startswith.
    """
    best = None
    for maildir, folder in pairs:
        if _within(path, root / maildir) and (
                best is None or len(maildir) > len(best[0])):
            best = (maildir, folder)
    return best


def origin_of(path, root, maildir):
    """The folder `path` sits in, relative to its ACCOUNT, without cur/new.

    The format MainWindow::originTagFor() writes and Restore reads:
    `acct/Inbox/new/x` is `Inbox`.
    """
    parts = path.parent.relative_to(root / maildir).parts
    if parts and parts[-1] in ("cur", "new"):
        parts = parts[:-1]
    return "/".join(parts)


def apply_moves(rules):
    """Move what each move rule matches into its account's role folder.

    True when done, including when there was nothing to do. False when a
    notmuch step failed, so the caller leaves tag:new in place. A message
    that cannot be moved (several files, no account, no folder for the role,
    a failed rename) is logged and left where it is; that is not a failure.
    """
    claimed = {}
    for rule in rules:
        if not rule.move:
            continue
        ids = search(mailrules.scoped_query(rule, SCOPE), "messages")
        if ids is None:
            return False
        for message_id in ids:
            if message_id in claimed:
                log(f"rule '{rule.id}': {message_id} already claimed by "
                    f"rule '{claimed[message_id].id}'")
                continue
            claimed[message_id] = rule

    if not claimed:
        return True

    root = mail_root()
    if root is None:
        log("no mail root; moving nothing")
        return False

    pairs = {role: qtmaildirconf.role_folders(role)
             for role in qtmaildirconf.MOVE_ROLES}

    to_tag = []
    renamed = False
    for message_id, rule in claimed.items():
        paths = search(f"id:{_quote(message_id)}", "files")
        if paths is None:
            return False
        if len(paths) != 1:
            log(f"rule '{rule.id}': {message_id} has {len(paths)} files; "
                f"not moved")
            continue

        source = Path(paths[0])
        account = account_of(source, root, pairs[rule.move])
        if account is None:
            log(f"rule '{rule.id}': {message_id} is in no account with a "
                f"{rule.move} folder; not moved")
            continue

        maildir, folder = account
        target = root / maildir / folder
        if _within(source, target):
            to_tag.append((message_id, rule.move, ""))
            continue

        # The configured folder must already exist. Creating it is how a
        # mistyped name reaches the server (item 103); cur/ inside a real
        # folder is safe to create.
        if not target.is_dir():
            log(f"rule '{rule.id}': {target} does not exist; not moved")
            continue

        origin = origin_of(source, root, maildir)
        try:
            (target / "cur").mkdir(exist_ok=True)
            os.rename(source, target / "cur" / fresh_name(source.name))
        except OSError as exc:
            log(f"rule '{rule.id}': cannot move {message_id}: {exc}")
            continue
        renamed = True
        to_tag.append((message_id, rule.move, origin))

    if renamed:
        # Records each rename against the message id it already has, so the
        # tags survive and no new.tags apply. --no-hooks, or this hook would
        # run itself.
        #
        # ponytail: this also indexes any file that landed after the outer
        # `notmuch new`, and the closing `-new` then consumes its marker
        # unseen by the rules. Under mailsync.sh mbsync has finished and the
        # lock is held, so nothing lands. If the hook ever runs beside a live
        # mbsync, scope the closing `-new` to the ids seen before this call.
        result = subprocess.run(["notmuch", "new", "--no-hooks", "--quiet"],
                                capture_output=True, text=True)
        if result.returncode != 0:
            log(f"notmuch new after moving failed: {result.stderr.strip()}")
            return False

    for message_id, role, origin in to_tag:
        # What Mark spam and Delete add, minus `unread`: the hook still never
        # removes that unattended (PROTECTED_REMOVALS).
        arguments = ["+spam" if role == "spam" else "+deleted", "-inbox"]
        if origin:
            arguments.append(f"+moved-from:{origin}")
        if not run_tag(arguments, f"id:{_quote(message_id)}"):
            return False

    log(f"moved {sum(1 for _, _, o in to_tag if o)} message(s), "
        f"tagged {len(to_tag)}")
    return True
```
